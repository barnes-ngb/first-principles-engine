namespace SpriteKind {
    export const Treasure = SpriteKind.create()
    export const CampFlag = SpriteKind.create()
}
sprites.onOverlap(SpriteKind.Player, SpriteKind.Treasure, function (player2, treasure) {
    treasure.destroy()
    info.changeScoreBy(pointsPerTreasure)
    music.play(music.melodyPlayable(music.baDing), music.PlaybackMode.InBackground)
    if (!(flagRaised) && info.score() >= flagGoal) {
        flagRaised = true
        flag.setImage(img`
            e 5 5 5 5 5 5 . 
            e 5 1 1 5 5 . . 
            e 5 1 5 5 . . . 
            e 5 5 5 5 5 . . 
            e 5 5 5 5 5 5 . 
            e . . . . . . . 
            e . . . . . . . 
            e . . . . . . . 
            e . . . . . . . 
            e e e . . . . . 
            `)
    }
})
function placeTreasure (column: number, row: number) {
    treasure = sprites.create(img`
        . . 4 4 4 4 . . 
        . 4 5 5 5 5 4 . 
        4 5 1 1 5 5 5 4 
        4 5 1 5 5 5 5 4 
        4 5 5 5 5 5 5 4 
        . 4 5 5 5 5 4 . 
        . . 4 4 4 4 . . 
        `, SpriteKind.Treasure)
    tiles.placeOnTile(treasure, tiles.getTileLocation(column, row))
}
let treasure: Sprite = null
let flagRaised = false
let flag: Sprite = null
let flagGoal = 0
let pointsPerTreasure = 0
let walkSpeed = 60
pointsPerTreasure = 1
flagGoal = 5
scene.setBackgroundColor(8)
tiles.setCurrentTilemap(tilemap`Lantern Cove`)
info.setScore(0)

    . . . c c c c . . . 
    . . c c . . c c . . 
    `, SpriteKind.Player)
tiles.placeOnTile(hero, tiles.getTileLocation(1, 1))
controller.moveSprite(hero, walkSpeed, walkSpeed)

    e . . . . . . . 
    e . . . . . . . 
    e . . . . . . . 
    e . . . . . . .
